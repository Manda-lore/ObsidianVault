# Two Pointers
## Leetcode 283: Move Zeros
```python
def move_zeros(arr):
    p1, p2 = 0, 0
    if len(arr) < 1:
        return arr
    while p2 < len(arr):
        if (arr[p1] == 0):
            while (p2 < len(arr)-1 and arr[p2] == 0):
                p2 += 1
            arr[p1], arr[p2] = arr[p2], arr[p1]
        p1 += 1
        p2 += 1
    return arr

def main():
    arr = [0, 1, 0, 3, 0, 12, 0]
    if len(arr) == 0:
        arr = list(map(int, input().split()))
    print(move_zeros(arr))

if __name__ == "__main__":
    main()
```
## Leetcode 11: Container with most water
```python
def container_with_most_water(arr):
    l,r = 0, len(arr)-1
    max_water, curr_water = 0, 0
    while l < r:
        curr_water = min(arr[l], arr[r]) * (r - l)
        max_water = max(max_water, curr_water)
        if arr[l] <= arr[r]:
            l += 1
        else:
            r -= 1
    return max_water

def main():
    arr = [1,8,6,2,5,4,8,3,7]
    if len(arr) == 0:
        arr = list(map(int, input().split()))
    print(container_with_most_water(arr))

if __name__ == "__main__":
    main()
```
# Sliding Window
## Leetcode 121: Best time to buy stocks
```python
def maximum_profit(arr):
    l,r = 0, 1
    max_profit, curr_profit = 0, 0
    while r < len(arr):
        curr_profit = arr[r] - arr[l]
        max_profit = max(curr_profit, max_profit)
        if (arr[l] > arr[r]):
            l = r
        r += 1
    return max_profit

def main():
    arr = [7,6,4,3,1]
    if len(arr) == 0:
        arr = list(map(int, input().split()))
    print(maximum_profit(arr))

if __name__ == "__main__":
    main()
```
